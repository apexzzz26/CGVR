
#include <GL/glut.h>
#include <iostream>
#include <algorithm>
using namespace std;

int n;
float x[100], y[100];

void drawLine(float x1, float y1, float x2, float y2) {
    glBegin(GL_LINES);
    glVertex2f(x1, y1);
    glVertex2f(x2, y2);
    glEnd();
}

void fill() {
    glColor3f(0, 0, 1);

    for (int s = 0; s <= 500; s++) {
        float a[100];
        int c = 0;

        for (int i = 0; i < n; i++) {
            int j = (i + 1) % n;

            float x1 = x[i], y1 = y[i];
            float x2 = x[j], y2 = y[j];

            if (y1 > y2) {
                swap(x1, x2);
                swap(y1, y2);
            }

            if (s >= y1 && s < y2) {
                a[c++] = x1 + (s - y1) * (x2 - x1) / (y2 - y1);
            }
        }

        sort(a, a + c);

        for (int i = 0; i + 1 < c; i += 2) {
            drawLine(a[i], s, a[i + 1], s);
        }
    }

    glFlush();
}

void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    fill();

    // Draw polygon boundary
    glColor3f(0, 1, 0);

    glBegin(GL_LINE_LOOP);
    for (int i = 0; i < n; i++) {
        glVertex2f(x[i], y[i]);
    }
    glEnd();

    glFlush();
}

int main(int argc, char** argv) {

    cout << "Enter number of vertices: ";
    cin >> n;

    if (n < 3 || n > 100) {
        cout << "Enter vertices between 3 and 100.";
        return 0;
    }

    cout << "Enter coordinates:\n";

    for (int i = 0; i < n; i++) {
        cin >> x[i] >> y[i];
    }

    glutInit(&argc, argv);

    glutInitDisplayMode(GLUT_RGB | GLUT_SINGLE);
    glutInitWindowSize(500, 500);

    glutCreateWindow("Scanline Fill");

    glClearColor(1, 1, 1, 1);

    gluOrtho2D(0, 500, 0, 500);

    glutDisplayFunc(display);

    glutMainLoop();

    return 0;
}
