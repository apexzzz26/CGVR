
#include <GL/glut.h>
#include <iostream>
using namespace std;

float xmin = -100, ymin = -100;
float xmax = 100, ymax = 100;

const int L = 8, R = 4, B = 2, T = 1;

int code(float x, float y) {
    int c = 0;

    if (x < xmin) c |= L;
    else if (x > xmax) c |= R;

    if (y < ymin) c |= B;
    else if (y > ymax) c |= T;

    return c;
}

void clip(float x1, float y1, float x2, float y2) {

    int c1 = code(x1, y1);
    int c2 = code(x2, y2);

    while (true) {

        if (!(c1 | c2))
            break;

        if (c1 & c2)
            return;

        int c = c1 ? c1 : c2;

        float x, y;

        if (c & T) {
            x = x1 + (x2 - x1) * (ymax - y1) / (y2 - y1);
            y = ymax;
        }
        else if (c & B) {
            x = x1 + (x2 - x1) * (ymin - y1) / (y2 - y1);
            y = ymin;
        }
        else if (c & R) {
            y = y1 + (y2 - y1) * (xmax - x1) / (x2 - x1);
            x = xmax;
        }
        else {
            y = y1 + (y2 - y1) * (xmin - x1) / (x2 - x1);
            x = xmin;
        }

        if (c == c1) {
            x1 = x;
            y1 = y;
            c1 = code(x1, y1);
        }
        else {
            x2 = x;
            y2 = y;
            c2 = code(x2, y2);
        }
    }

    glColor3f(0, 0, 1);

    glBegin(GL_LINES);
    glVertex2f(x1, y1);
    glVertex2f(x2, y2);
    glEnd();
}

void display() {

    glClear(GL_COLOR_BUFFER_BIT);

    // Original lines in red
    glColor3f(1, 0, 0);

    glBegin(GL_LINES);
    glVertex2f(-120, 160);
    glVertex2f(160, 80);

    glVertex2f(60, 30);
    glVertex2f(90, 120);
    glEnd();

    // Clipping window in green
    glColor3f(0, 1, 0);

    glBegin(GL_LINE_LOOP);
    glVertex2f(xmin, ymin);
    glVertex2f(xmax, ymin);
    glVertex2f(xmax, ymax);
    glVertex2f(xmin, ymax);
    glEnd();

    // Clipped lines in blue
    clip(-120, 160, 160, 80);
    clip(60, 30, 90, 120);

    glFlush();
}

int main(int argc, char** argv) {

    glutInit(&argc, argv);

    glutInitDisplayMode(GLUT_RGB | GLUT_SINGLE);
    glutInitWindowSize(500, 500);

    glutCreateWindow("Cohen Sutherland Line Clipping");

    glClearColor(1, 1, 1, 1);

    gluOrtho2D(-250, 250, -250, 250);

    glutDisplayFunc(display);

    glutMainLoop();

    return 0;
}
