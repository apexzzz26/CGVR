
#include <GL/glut.h>
#include <iostream>
#include <cmath>
using namespace std;

float t, xr, yr, m, c;

float h[11][2] = {
    {100,100}, {140,100}, {140,150},
    {160,150}, {160,100}, {200,100},
    {200,200}, {220,200}, {150,300},
    {80,200}, {100,200}
};

void house() {
    glBegin(GL_LINE_LOOP);

    for (int i = 0; i < 11; i++) {
        glVertex2fv(h[i]);
    }

    glEnd();
}

// Rotation
void rotation() {
    glClear(GL_COLOR_BUFFER_BIT);

    // Original house - Red
    glColor3f(1, 0, 0);
    house();

    glPushMatrix();

    glTranslatef(xr, yr, 0);
    glRotatef(t, 0, 0, 1);
    glTranslatef(-xr, -yr, 0);

    // Rotated house - Blue
    glColor3f(0, 0, 1);
    house();

    glPopMatrix();

    glFlush();
}

// Reflection
void reflection() {
    glClear(GL_COLOR_BUFFER_BIT);

    // Original house - Red
    glColor3f(1, 0, 0);
    house();

    // Draw reflection line y = mx + c
    glColor3f(0, 1, 0);

    glBegin(GL_LINES);
    glVertex2f(0, c);
    glVertex2f(400, 400 * m + c);
    glEnd();

    float r = atan(m) * 180.0 / 3.14159265;

    glPushMatrix();

    glTranslatef(0, c, 0);
    glRotatef(r, 0, 0, 1);
    glScalef(1, -1, 1);
    glRotatef(-r, 0, 0, 1);
    glTranslatef(0, -c, 0);

    // Reflected house - Blue
    glColor3f(0, 0, 1);
    house();

    glPopMatrix();

    glFlush();
}

int main(int argc, char** argv) {

    int op;

    cout << "1. Rotation  2. Reflection: ";
    cin >> op;

    if (op == 1) {
        cout << "Enter angle, xr, yr: ";
        cin >> t >> xr >> yr;
    }
    else if (op == 2) {
        cout << "Enter m and c: ";
        cin >> m >> c;
    }
    else {
        cout << "Invalid choice!";
        return 0;
    }

    glutInit(&argc, argv);

    glutInitDisplayMode(GLUT_RGB | GLUT_SINGLE);
    glutInitWindowSize(500, 500);

    glutCreateWindow("House Rotation and Reflection");

    glClearColor(1, 1, 1, 1);

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(0, 500, 0, 500);

    glMatrixMode(GL_MODELVIEW);
    glLoadIdentity();

    glutDisplayFunc(op == 1 ? rotation : reflection);

    glutMainLoop();

    return 0;
}
